import collections, os
from pathlib import Path
import nibabel as nib

root = Path(os.environ.get('TASK1_DATA_ROOT', '/2026aicompetition/datasets/training'))
comp = root / 'annotation' / 'Composition'
print('Composition 目录:', comp, '存在:', comp.is_dir())

files = [p for p in comp.rglob('*') if p.is_file()]
print('文件总数:', len(files))
print('扩展名分布:', dict(collections.Counter(p.suffix.lower() for p in files).most_common(10)))
print('无扩展名文件数:', sum(1 for p in files if not p.suffix))
print('目录深度示例:', [str(p.relative_to(comp)) for p in files[:5]])

HINTS = ('mask', 'seg', 'label', 'roi')
nii = [p for p in files if p.name.lower().endswith(('.nii', '.nii.gz'))]
kept = [p for p in nii if not any(h in p.name.lower() for h in HINTS)]
print('NIfTI 文件数:', len(nii), '| 通过名称过滤的:', len(kept))
if len(nii) != len(kept):
    print('  被 mask/seg/label/roi 过滤掉的前几个:', [p.name for p in nii if p not in kept][:5])

print('--- 试读前 3 个 NIfTI ---')
for p in nii[:3]:
    try:
        img = nib.load(str(p))
        print('  OK  ', p.name, img.shape, img.get_data_dtype())
    except Exception as exc:
        print('  FAIL', p.name, type(exc).__name__, str(exc)[:120])

print('--- 前 3 个文件的魔数（判断真实格式）---')
for p in files[:3]:
    head = p.read_bytes()[:160]
    print(' ', p.name, '| 开头:', head[:4].hex(), '| 偏移128:', head[128:132])
